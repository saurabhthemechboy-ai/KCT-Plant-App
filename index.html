function apiCall(action, payload, onSuccess, onFail) { onSuccess = onSuccess || function () {}; onFail = onFail || function (err) { alert("Network error: " + (err && err.message ? err.message : err)); }; if (typeof google !== "undefined" && google.script && google.script.run) { const args = buildArgsForAction_(action, payload); const runner = google.script.run .withSuccessHandler(onSuccess) .withFailureHandler(onFail); runner[action].apply(runner, args); return; } const WORKER_URL = "https://kct-api.saurabhthemechboy.workers.dev"; fetch(WORKER_URL + "/" + encodeURIComponent(action), { method: "POST", headers: { "Content-Type": "text/plain;charset=utf-8" }, body: JSON.stringify(payload || {}) }) .then(function (r) { if (!r.ok) throw new Error("HTTP " + r.status); return r.text(); }) .then(function (text) { let res; try { res = JSON.parse(text); } catch (e) { const m = text.match(/\{[\s\S]*\}/); if (m) res = JSON.parse(m[0]); else throw new Error("Invalid response from server"); } if (res && res.error) onFail(new Error(res.error)); else onSuccess(res); }) .catch(function (err) { console.error("[KCT API] fetch error:", err); onFail(err); }); }
